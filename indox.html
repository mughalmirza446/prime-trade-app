// In Google Sheets: Extensions → Apps Script // Paste this code, then Deploy → New deployment → Web app // Execute as: Me • Who has access: Anyone // Paste the URL you get below // Yeh script DONO kaam karti hai: // 1) Har nayi entry ki ROW alag sheet mein (New Cash Memo, Add New Product, // Add New Wholesaler, New Expense, New Reinvestment) - app khud bhejti hai // 2) Poora data SNAPSHOT sheet mein - app ke Upload/Download buttons ke liye function doPost(e){ var ss=SpreadsheetApp.getActiveSpreadsheet(); var d=JSON.parse(e.postData.contents); if(d.sheet){ // === Nayi entry ki row us sheet mein likho (sheet na ho to bana do) === var sh=ss.getSheetByName(d.sheet); if(!sh){sh=ss.insertSheet(d.sheet);} var keys=Object.keys(d.data||{}); if(keys.length===0){ return ContentService.createTextOutput(JSON.stringify({status:"error",message:"empty"})).setMimeType(ContentService.MimeType.JSON); } if(sh.getLastRow()===0){ sh.getRange(1,1,1,keys.length+1).setValues([["Saved At"].concat(keys)]); } var row=[new Date(d.id||Date.now())].concat(keys.map(function(k){return d.data[k];})); sh.appendRow(row); return ContentService.createTextOutput(JSON.stringify({status:"success",sheet:d.sheet})).setMimeType(ContentService.MimeType.JSON); } // === Purana snapshot system (Upload/Download ke liye) === var snap=ss.getSheetByName("SNAPSHOT"); if(!snap){snap=ss.insertSheet("SNAPSHOT");} snap.clearContents(); snap.getRange(1,1).setValue(JSON.stringify(d)); return ContentService.createTextOutput(JSON.stringify({status:"success"})).setMimeType(ContentService.MimeType.JSON); } function doGet(){ var ss=SpreadsheetApp.getActiveSpreadsheet(); var snap=ss.getSheetByName("SNAPSHOT")||ss.getSheets()[0]; var v=snap.getRange(1,1).getValue(); return ContentService.createTextOutput(v||"{}").setMimeType(ContentService.MimeType.JSON); }

Google Sheets Web App URL pehle se connected hai. Har nayi memo/product/wholesaler/expense/reinvestment ki row Google Sheet mein save hogi; internet na ho to offline queue mein rahegi aur online aate hi sync ho jayegi. Upload/Download buttons bhi isi URL ke SNAPSHOT system ke sath kaam karte hain.
